cmake_minimum_required(VERSION 3.16)
project(Knurl VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(Threads REQUIRED)
file(GLOB KNURL_SOURCES CONFIGURE_DEPENDS src/*.c)
add_executable(knurl ${KNURL_SOURCES})
target_include_directories(knurl PRIVATE include)
target_compile_definitions(knurl PRIVATE _POSIX_C_SOURCE=200809L)
target_compile_options(knurl PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(knurl PRIVATE Threads::Threads m)
