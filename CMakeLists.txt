cmake_minimum_required(VERSION 3.16)
project(Knurl VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_executable(knurl src/knurl.c)
add_executable(knurl-check src/check_main.c src/check_common.c src/check_cpu.c src/check_memory.c src/check_network.c src/check_disk.c)
foreach(target knurl knurl-check)
  target_include_directories(${target} PRIVATE include)
  target_compile_definitions(${target} PRIVATE _POSIX_C_SOURCE=200809L)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(${target} PRIVATE Threads::Threads m)
endforeach()
